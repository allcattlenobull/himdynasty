<script>
(function () {
    "use strict";

    const body = document.getElementById("bigBoardBody");
    const resetButton = document.getElementById("resetSort");
    const sortButtons = Array.from(document.querySelectorAll("button[data-sort-key]"));
    const desktopSortButtons = Array.from(document.querySelectorAll(".sort-button"));
    const navToggle = document.getElementById("navToggle");
    const navLinks = document.getElementById("navLinks");

    function compareText(a, b) {
        return a.localeCompare(b, undefined, { numeric: true, sensitivity: "base" });
    }

    function updateSortIndicators(key, direction) {
        desktopSortButtons.forEach(function (button) {
            const header = button.closest("th");
            if (button.dataset.sortKey === key) {
                button.dataset.direction = direction;
                header.setAttribute("aria-sort", direction === "asc" ? "ascending" : "descending");
            } else {
                button.removeAttribute("data-direction");
                header.setAttribute("aria-sort", "none");
            }
        });
    }

    function sortRows(key, forcedDirection) {
        const currentKey = body.dataset.sortKey;
        const currentDirection = body.dataset.sortDir || "asc";
        const direction = forcedDirection || (currentKey === key && currentDirection === "asc" ? "desc" : "asc");
        const rows = Array.from(body.querySelectorAll("tr"));

        rows.sort(function (rowA, rowB) {
            const rankA = Number(rowA.dataset.rank) || 0;
            const rankB = Number(rowB.dataset.rank) || 0;

            if (key === "rank") {
                return direction === "asc" ? rankA - rankB : rankB - rankA;
            }

            const valueA = (rowA.dataset[key] || "").trim();
            const valueB = (rowB.dataset[key] || "").trim();
            const primary = compareText(valueA, valueB);

            if (primary === 0) return rankA - rankB;
            return direction === "asc" ? primary : -primary;
        });

        rows.forEach(function (row) { body.appendChild(row); });
        body.dataset.sortKey = key;
        body.dataset.sortDir = direction;
        updateSortIndicators(key, direction);
    }

    sortButtons.forEach(function (button) {
        button.addEventListener("click", function () {
            sortRows(button.dataset.sortKey);
        });
    });

    resetButton.addEventListener("click", function () {
        sortRows("rank", "asc");
    });

    navToggle.addEventListener("click", function () {
        const isOpen = navLinks.classList.toggle("open");
        navToggle.setAttribute("aria-expanded", isOpen ? "true" : "false");
    });
})();
</script>
